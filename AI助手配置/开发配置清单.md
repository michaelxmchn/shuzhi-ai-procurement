## 一、开发概述

### 1.1 系统架构

```
用户 → 全能助手 → 需求采集 → 分类判断 → 专业助手 → 执行层
                                            ↓
                                        搜索/询价
```

| 要求 | 说明 |
|------|------|
| **框架** | Next.js 14 + React + Tailwind CSS |
| **后端** | FastAPI + Python |
| **AI模型** | DeepSeek VL + Ollama 本地模型 |
| **数据库** | PostgreSQL + Redis |
| **搜索** | Brave Search API |

### 4.2 核心组件代码

#### ChatInterface.jsx

```jsx
import React, { useState, useEffect, useRef } from 'react';
import MessageBubble from './MessageBubble';
import InputArea from './InputArea';
import useChat from '../hooks/useChat';

export default function ChatInterface() {
    const [messages, setMessages] = useState([]);
    const [assistantType, setAssistantType] = useState('general');
    const [isTyping, setIsTyping] = useState(false);
    const messagesEndRef = useRef(null);

    const { sendMessage, determineAssistant } = useChat();

    // 发送消息
    const handleSend = async (text) => {
        // 添加用户消息
        setMessages(prev => [...prev, {
            role: 'user',
            content: text,
            timestamp: new Date()
        }]);

        setIsTyping(true);

        // 判断是否需要转接专业助手
        const assistant = await determineAssistant(text);
        if (assistant.type !== assistantType && assistant.confidence > 0.8) {
            setAssistantType(assistant.type);
            // 添加转接提示
            setMessages(prev => [...prev, {
                role: 'system',
                content: `已为您转接${assistant.type}专家...`,
                timestamp: new Date()
            }]);
        }

        // 发送消息
        const response = await sendMessage(text, assistantType);
        
        setIsTyping(false);
        setMessages(prev => [...prev, {
            role: 'assistant',
            content: response.response,
            timestamp: new Date()
        }]);
    };

    return (
        <div className="chat-container">
            <div className="messages">
                {messages.map((msg, index) => (
                    <MessageBubble 
                        key={index}
                        role={msg.role}
                        content={msg.content}
                        timestamp={msg.timestamp}
                    />
                ))}
                {isTyping && <TypingIndicator />}
                <div ref={messagesEndRef} />
            </div>
            <InputArea onSend={handleSend} />
        </div>
    );
}
```

### 5.2 提示词加载示例

```python
class PromptManager:
    def __init__(self, assistant_type):
        self.assistant_type = assistant_type
        self.prompts = self._load_prompts(assistant_type)
    
    def _load_prompts(self, assistant_type):
        prompts = {}
        base_path = f"prompts/{assistant_type}"
        
        for file in os.listdir(base_path):
            if file.endswith('.md'):
                with open(f"{base_path}/{file}") as f:
                    prompts[file.replace('.md', '')] = f.read()
        
        return prompts
    
    def get_system_prompt(self):
        return self.prompts.get('system', '')
    
    def get_response(self, user_message, history):
        # 组合提示词
        system_prompt = self.get_system_prompt()
        context = self._format_history(history)
        user_query = f"{system_prompt}\n\n{context}\n\n用户: {user_message}"
```

```python
import pytest
from app.assistants import GeneralAssistant, ElectronicsAssistant

def test_general_opening():
    assistant = GeneralAssistant()
    response = assistant.respond("你好")
    assert "您好" in response
    assert "采购助手" in response

def test_electronics_transfer():
    assistant = GeneralAssistant()
```
